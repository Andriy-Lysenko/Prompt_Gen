try:
    response = requests.post(
        f"{fooocus_url}/run/predict",
        json={
            "data": [prompt]  # Текстовый запрос
        }
    )
    
    if response.status_code == 200:
        # Получаем изображение из ответа
        result = response.json()
        st.write("Response JSON:", result)  # Логирование ответа
        image_data = result["data"][0]  # Изображение в формате base64
        
        # Декодируем изображение
        img_bytes = base64.b64decode(image_data.split(",")[1])
        image = Image.open(BytesIO(img_bytes))
        
        # Показываем изображение
        st.subheader("Generated Scene")
        st.image(image, caption=prompt, use_column_width=True)
    else:
        st.error(f"Failed to generate image. Status code: {response.status_code}")
        st.write("Response text:", response.text)  # Логирование текста ответа
except Exception as e:
    st.error(f"An error occurred: {e}")
